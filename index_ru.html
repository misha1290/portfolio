Beauty Lab

Vanilla JS • Tailwind CSS • Fetch API

Премиальный двуязычный (RU/DE) лендинг, разработанный для эстетического центра в Германии. Проект включает динамический рендеринг DOM, анимации при скролле и бессерверную систему сбора заявок с интеграцией в Telegram.

  • Производительность: Стилизовано строго с помощью Tailwind CSS и анимировано через нативный IntersectionObserver API для оптимальной скорости рендеринга.
telegram-api.js 
// Asynchronous lead capture & CORS handling
        const GOOGLE_APP_URL = 'https://script.google.com/macros/s/.../exec'; 

        async function sendToTelegram(message, inputIdsToClear) {
            try {
                const response = await fetch(GOOGLE_APP_URL, {
                    method: 'POST',
                    // Using text/plain to avoid CORS preflight issues in Google Script
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify({ message: message })
                });
                
                const result = await response.json();
                
                if (result.status === 'success') {
                    showSuccessModal();
                    clearFormStorage(inputIdsToClear); // Clears localStorage upon success
                } else {
                    console.error('Server Error:', result.message);
                }
            } catch (error) {
                console.error('Network Error:', error);
            }
        }

Time Sync Manager & Shop

JavaScript (ES6+) • Fetch API • LocalStorage

Веб-приложение с таймером обратного отсчета, синхронизированным с глобальным API времени. Включает виртуальный магазин, где пользователи могут покупать и использовать усилители, влияющие на время.

Ключевые особенности:

  • Постоянное управление состоянием: Использование localStorage для сохранения времени окончания таймера, моментов паузы и инвентаря магазина между перезагрузками браузера.
  • Динамические обновления UI: Динамический расчет часов, минут и секунд, а также очистка интервала при достижении таймером нуля.
timer-sync.js 
// Fetching real UTC time to prevent client-side cheating
        async function getServerTime() {
        try {
            const response = await fetch("https://worldtimeapi.org/api/ip");
            const data = await response.json();
            return new Date(data.utc_datetime);
        } catch (error) {
            console.error("API Error:", error);
            return new Date(); // fallback: local time
        }
        }

radio-drag.js 
// Handling draggable radio slider and track switching
        draggable.onmousedown = function(event) {
            isDragging = true;
            document.onmousemove = function(event) {
                if (isDragging) {
                    const containerRect = container.getBoundingClientRect();
                    let newLeft = event.clientX - containerRect.left - draggable.offsetWidth / 2;

                    let closestPosition = positions[0];
                    let closestDistance = Math.abs(newLeft - positions[0]);
                    for (let pos of positions) {
                        let distance = Math.abs(newLeft - pos);
                        if (distance < closestDistance) {
                            closestPosition = pos;
                            closestDistance = distance;
                        }
                    }
                    draggable.style.left = closestPosition + 'px';
                }
            };
        };
